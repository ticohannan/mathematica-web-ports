
### [Title]
Paths inside a Polygon

### [Section]
Caption<<PaneSelectorBox>>

### [Text]
\<This Demonstration illustrates an algorithm for finding the shortest path that stays inside a polygon and connects two given interior points. Aside from the start and finish, such a path must go from reflex vertex to reflex vertex; thus you start by first making the graph whose edges (shown in blue) are all the segments that stay inside the polygon and connect two such vertices (and the start and finish points). Then a standard shortest path algorithm yields the desired path. You can drag the start and finish points to new locations.\>

SignedArea[poly_]:=Plus@@Flatten[poly*Map[{1,-1}*Reverse[#]&,RotateLeft[poly]]]/2/;Length[poly]>=3;

AreaSW[pp_]:=Abs[SignedArea[pp]];

IntersectOpen[{a_,b_},{c_,d_}]:=((LeftOf[a,b,c]&&RightOf[a,b,d])||(LeftOf[a,b,d]&&RightOf[a,b,c]))&&((LeftOf[c,d,a]&&RightOf[c,d,b])||(LeftOf[c,d,b]&&RightOf[c,d,a]));

BetweenSW[p_,q_,r_]:=Orientation[{p,q,r}]==0&&If[p[[1]]!=r[[1]],p[[1]]<=r[[1]]<=q[[1]]||q[[1]]<=r[[1]]<=p[[1]],p[[2]]<=r[[2]]<=q[[2]]||q[[2]]<=r[[2]]<=p[[2]]];

IntersectClosed[{a_,b_},{c_,d_}]:=IntersectOpen[{a,b},{c,d}]||BetweenSW[a,b,c]||BetweenSW[a,b,d]||BetweenSW[c,d,a]||BetweenSW[c,d,b];

Orientation[triangle_] := Sign[SignedArea[triangle]] /; Length[triangle] == 3;

ConvexVertex[pp_]:= Ordering[pp, 1][[1]];

Orientation[poly_]:=Orientation[
     poly[[ConvexVertex[poly]+{-1,0,1}/.{
            0 -> Length[poly],Length[poly]+1->1}
            ]]] /; Length[poly]>3;

LeftOf[p_,q_,r_] := Orientation[{p,q,r}]==1;RightOf[p_,q_,r_]:=(Orientation[{p,q,r}]==-1);

Inside[{p_,q_,r_},pt_]:=Module[{pp,qq,rr},{pp,qq,rr}=If[Orientation[{p,q,r}]==-1,Reverse,Identity][{p,q,r}];
(!LeftOf[qq,pp,pt]&&!LeftOf[rr,qq,pt]&&!LeftOf[pp,rr,pt])];

Inside[p_List/;Length[p]>3,pt_List]:=Module[{tris=(p[[#1]]&)/@Triangulate[p],ans=False,ii=0},While[!ans&&ii<Length[tris],ii++;ans=Inside[tris[[ii]],pt]];ans];

PolygonDiagonal[poly_]:=Module[{v=ConvexVertex[poly],invertices,u,w},u=v-1/.0->Length[poly];
w=v+1/.1+Length[poly]->1;
invertices=Select[Delete[poly,{{u},{v},{w}}],Inside[poly[[{u,v,w}]],#]&];
Sort[If[invertices=={},{u,w},Identity[{Position[poly,Last[Sort[invertices,AreaSW[{poly[[u]],#1,poly[[w]]}]<=AreaSW[{poly[[u]],#2,poly[[w]]}]&]]][[1,1]],v}]]]];

Triangulate[poly_List]:=If[Orientation[poly]==1,Triangulate[poly,Range[Length[poly]]],Reverse/@(Length[poly]+1-Triangulate[Reverse[poly],Range[Length[poly]]])];

Triangulate[poly_,inds_]:=Module[{d=PolygonDiagonal[poly[[inds]]]},Join[Triangulate[poly,Take[inds,d]],Triangulate[poly,Drop[inds,d+{1,-1}]]]];

Triangulate[_,{i1_,i2_,i3_}]:={{i1,i2,i3}};

extend[a_]:=Flatten[{a,{First[a]}},1];

ReflexVertexQ[poly_,ii_] := !LeftOf[poly[[ii-1/. 0-> Length[poly]]],poly[[ii]],poly[[ii+1/.  1+ Length[poly]->1]]];
InsideFast[tris_, pt_] := (ans1=False; Do[If[Inside[tt, pt],ans1=True;Break[]], {tt, tris}]; Return[ans1]);

options[ShortestInsidePath] = {AlgorithmTraceQ->False, ShowDistance->True};
VisibleReflexEdges//Clear;
VisibleReflexEdges[poly_] :=VisibleReflexEdges[poly]= Module[{i,j,n,v,tris, verts, border, borderReflexEdges, possibleVVEdges, killer},
n=Length[poly];
v=If[Orientation[poly]==-1, Reverse, Identity][poly];
tris = v[[#]]&/@ Triangulate[v];
vertsShow=verts =Select[ Range[n], ReflexVertexQ[v,#]&];
border = Partition[Append[Range[n],1],2,1];
 borderReflexEdges =Select[border, MemberQ[verts,#[[1]]]&& MemberQ[verts,#[[2]]] &];
possibleVVEdges=Flatten[Table[verts[[{i,j}]], {i,Length[verts]-(3;1)}, {j, i+1,Length[verts]-(2;0)}],1];
possibleVVEdges=Select[possibleVVEdges, InsideFast[tris,Mean[v[[#]]]]&];
Do[killer=v[[border[[i]]]]; possibleVVEdges=Select[possibleVVEdges ,MemberQ[border[[i]], #[[1]]]||MemberQ[border[[i]], #[[2]]]|| !IntersectClosed[v[[#]],killer]&],{i,1, n}];
Union[borderReflexEdges, possibleVVEdges]];

ShortestInsidePath[s_, start_, finish_, opts___] :=Module[{mPrint,n,ss,v,tris, trQ, distQ, verts, border, finishEdges, killer,startEdges, pp, g1, sp, dist}, n=Length[s];
{distQ,trQ} = {ShowDistance,AlgorithmTraceQ}/. {opts} /. options[ShortestInsidePath];

If[trQ, mPrint = Print, mPrint=mPrint];
ss=If[Orientation[s]==-1, Reverse, Identity][s];
v=Join[ss, {start, finish}];
tris = v[[#]]&/@ Triangulate[ss];
If[!InsideFast[tris, start], Print["\<The starting point is outside the polygon\>"]; Return[HoldForm[ShortestInsidePath[s, start, finish, opts]]]];

If[!InsideFast[tris, finish], Print["\<The ending point is outside the polygon\>"]; Return[HoldForm[ShortestInsidePath[s, start, finish, opts]]]];

verts =Join[Select[ Range[n], ReflexVertexQ[ss,#]&], {n+1, n+2}];
mPrint[Length[verts],  "\< = number of reflex verts; now triangulating\>"];

mPrint["\<got triangulation\>"];
border = Partition[Append[Range[n],1],2,1];
(* three types of legal moves   type 1: border edge from reflex to reflex   type 2: vertex to vertex, not an edge
*) 

mPrint["\<starting to kill vertex to vertex\>"];
possibleVVEdges=VisibleReflexEdges[ss];
mPrint["\<finished kill vertex to vertex\>"];
startEdges = Select[Table[verts[[{i,-2}]],{i,1,Length[verts]-2} ],(   InsideFast[tris,Mean[v[[#]]]]) &];

If[InsideFast[tris,Mean@v[[{n+1,n+2}]]],startEdges=Append[ Select[startEdges, InsideFast[tris,Mean[v[[#]]]]&], {n+1,n+2}]];
mPrint["\<start \>", startEdges];Do[killer=v[[border[[i]]]]; startEdges=Select[startEdges ,MemberQ[v[[border[[i]]]], v[[#[[1]]]]]||MemberQ[v[[border[[i]]]],v[[ #[[2]]]]]||!IntersectClosed[v[[#]],killer]&],{i,1, n}];
mPrint["\<start\>", startEdges];

finishEdges = Select[Table[verts[[{i,-1}]],{i,1,Length[verts]-2} ] ,(   InsideFast[tris,Mean[v[[#]]]]) &];
finishEdges= Select[finishEdges, InsideFast[tris,Mean[v[[#]]]]&];
Do[killer=v[[border[[i]]]]; finishEdges=Select[finishEdges ,MemberQ[v[[border[[i]]]], v[[#[[1]]]]]||MemberQ[v[[border[[i]]]], v[[#[[2]]]]]||!IntersectClosed[v[[#]],killer]&],{i,1, n}];

mPrint["\<Forming graph\>"];
pp=Union[ possibleVVEdges, startEdges, finishEdges];
mPrint[pp];
(*g=Graph[List/@pp, List/@ v];*)
g1 =Graph[v, pp/. {i_Integer,j_} :>i\[UndirectedEdge]j, EdgeWeight->Table[Norm[v[[e[[1]]]]-v[[e[[2]]]]], {e, pp}]];
mPrint[g1];
mPrint[VertexCount[g1]];

mPrint["\<Finding shortest path\>"];
