
### [Subchapter]
\<Demonstration of Attractive, Repulsive, and Combination Potential Fields\>

### [Input] (init)
Uatt[x_,y_,ζ_,d_] :=Module[{dist = Sqrt[x^2+y^2] },If[dist<= d, (1)/(2)ζ (x^2+y^2) , d ζ (dist-(1)/(2) d)]];
Urep[x_,y_,xo_,yo_,η_,ρ0_] :=Module[{dist =Sqrt[(x-xo)^2+(y-yo)^2] },If[dist<= ρ0, (1)/(2)η((1)/(dist)-(1)/(ρ0))^2 ,0]];
