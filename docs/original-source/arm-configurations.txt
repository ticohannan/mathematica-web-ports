### [Title]
Common Robot Arm Configurations

### [Section]
Caption<<PaneSelectorBox>>

### [Text]
\<Robot manipulators can be concisely described by a convention called the Denavit\[Dash]Hartenberg parameters (DH parameters) to assign one variable and three parameters to describe the distance and angular offset between each joint. This Demonstration allows you to choose an arbitrary robot specified by DH parameters and animate the degrees of freedom.\>

{Blue,Rotate[drawZArrow[jr],π/2,{0,1,0}]},
{Green,Rotate[drawZArrow[jr],-π/2,{1,0,0}]}}

### [Input] (init)
drawJoint[ j_,d_,r_,Theta_]:=Module[{jr = 1/5,ar = 1/20,pr=1/7,vr=1/6},{
(*draw coordinate axis*)
drawCoordAxes[jr],
Opacity[1],
(*draw z axis*)
{Opacity[0.5],Gray,If[MemberQ[{"\<Prismatic\>","\<prismatic\>","\<P\>","\<p\>"},j],Cuboid[{-ar,-ar,-1+d-jr-.01},{ar,ar,d+.01}],Cylinder[{{0,0,Min[-ar,d-jr]-.01},{0,0,Max[ar,d]+.01}},ar]]},
(*draw joint*)
{LightBlue,If[j=="\<p\>",
{Cuboid[{-jr,-jr,-jr},{jr,jr,+jr-.1}],
Cuboid[{-jr,-jr,+jr},{jr,jr,+jr+.05}]
},
{Cylinder[{{0,0,-jr-.05},{0,0,+jr+.05}},.7*jr]
}]},
(*draw x(i+1) *)
Rotate[{Opacity[0.5],Gray,Cuboid[{-ar,-ar,d-ar},{r,ar,d+ar}]},Theta,{0,0,1}]
}]

### [Input] (init)
drawShaft[ j_,d_,r_,Theta_]:=Module[{jr = 1/5,ar = 1/20},{
Opacity[1],
(*draw z axis*)
{Opacity[0.5],Gray,If[j=="\<p\>" ,Cuboid[{-ar,-ar,-1+d-jr-.01},{ar,ar,d+.01}],Cylinder[{{0,0,Min[-ar,d-jr]-.01},{0,0,Max[ar,d]+.01}},ar]]}
}]

### [Input] (init)
drawGripper[g_,r_]:= Module[{jr = 1/5,ar = 1/20},{
(*draw z axis*)
drawCoordAxes[jr],
If[r!= 0,
{Gray,Cuboid[{-2ar,-ar,-4ar},{0,ar,4ar}],
Cuboid[{0ar,-ar,g 2ar},{4ar,ar,2(1+g)ar}],
Cuboid[{0ar,-ar,-g 2ar},{4ar,ar,-2(1+g)ar}]},
(*looks better if hand points along d*)
{Gray,
Cuboid[{-4ar,-ar,-2ar},{4ar,ar,0}],
Cuboid[{g 2ar,-ar,0ar},{2(1+g)ar,ar,4ar}],
Cuboid[{-g 2ar,-ar,0ar},{-2(1+g)ar,ar,4ar}]}]
}]

### [Input] (init)
(*math function to create a T matrix*)
dhTransform[d_,r_,Theta_,Alpha_]:=RotationTransform[Theta,{0,0,1}].TranslationTransform[{0,0,d}].TranslationTransform[{r,0,0}].RotationTransform[Alpha,{1,0,0}]

### [Input] (init)
makeDHControl[unusedVar_,dhp_]:=Module[{jointtype,i,a,theta,alpha,d,jointnum,dof,dhdim},
dhdim =Dimensions[dhp];dof=dhdim[[2]];
a=ConstantArray[0,dof+1];theta=ConstantArray[0,dof+1];jointtype = ConstantArray[0,dof+1];
d=ConstantArray[0,dof+1];alpha=ConstantArray[0,dof+1];jointnum=Join[{"\<joint\>"},Array[#&,dof]];
alpha[[1]]="\<Alpha\>";d[[1]]="\<d\>";jointtype[[1]]="\<type\>";a[[1]]="\<r\>";
Table[{jointtype[[i+1]]= ToString[dhp[[1,i]]],a[[i+1]]=dhp[[2,i]],
alpha[[i+1]]=dhp[[3,i]],d[[i+1]]=dhp[[4,i]],theta[[i+1]]=dhp[[5,i]]},{i,1,dof}];
Table[If[dhp[[1,i]]== "\<r\>",theta[[i+1]]=Subsuperscript["\<Theta\>",i,"\<*\>"],d[[i+1]]=Subsuperscript["\<d\>",i,"\<*\>"] ],{i,1,dof}];
Text[Grid[Transpose[{jointnum,jointtype,a,alpha,d,theta}],Frame->All]]]

Manipulate[
Module[{jr = 1/10,ar = 1/40,Ad,Td,i,DHdim,a,d,theta,alpha,jointtype},
DHdim =Dimensions[DHParameters];
dof=DHdim[[2]];
Table[{jointtype[i]= ToString[DHParameters[[1,i]]],a[i]=DHParameters[[2,i]],alpha[i]=DHParameters[[3,i]],d[i]=DHParameters[[4,i]],theta[i]=DHParameters[[5,i]]},{i,1,DHdim[[2]]}];

Ad =Table[ If[
MemberQ[{"\<Prismatic\>","\<prismatic\>","\<P\>","\<p\>"},ToString[DHParameters[[1,i]]]],
dhTransform[params[[i]],a[i],theta[i],alpha[i]],
dhTransform[d[i],a[i],params[[i]],alpha[i]]
],{i,1,DHdim[[2]]}];
Td[1] = Ad[[1]];
Table[Td[i] = Chop[Td[i-1].Ad[[i]]],{i,2,DHdim[[2]]}];
Column[{
Graphics3D[{
(*ground*)
{LightBrown,Cylinder[{{0,0,-2/5},{0,0,-1/5-1/20}},2.2]},
If[MemberQ[{"\<Revolute\>","\<revolute\>","\<R\>","\<r\>"},jointtype[1]],
drawJoint[jointtype[1],d[1],a[1],params[[1]]],drawJoint[jointtype[1],params[[1]],a[1],theta[1]]],
If[dof==1,GeometricTransformation[drawGripper[g,0],Chop[Td[dof]]],

 Table[If[MemberQ[{"\<Revolute\>","\<revolute\>","\<R\>","\<r\>"},jointtype[i]],
GeometricTransformation[drawJoint[jointtype[i],d[i],a[i],params[[i]]],Td[i-1]],
GeometricTransformation[drawJoint[jointtype[i],params[[i]],a[i],theta[i]],Td[i-1]]],{i,2,dof}]],
If[MemberQ[{"\<Revolute\>","\<revolute\>","\<R\>","\<r\>"},jointtype[i]],
GeometricTransformation[drawShaft[ jointtype[dof],d[dof], a[dof],params[[dof]]],Td[dof-1]],

],

]],{i,dof}](*Table*)]],(*Dynamic*)
"\<\>",
Control@{{g,1,"\<grip\>"},0,1,0.01,ImageSize->Small,Appearance->"\<Labeled\>"},

### [Text]
\<This Demonstration creates a serial-link robot arm with some common configurations. You can manipulate each joint using the sliders.\>

### [Text]
\<Degrees of freedom (DOF) is the number of joints that the robot has. The DOF is equal to the dimension of the robot's configuration space.\>

### [Text]
The DH parameters are link length r, link twist Alpha, link offset d and joint angle Theta. These four parameters are associated with link i and joint i.
